ШАГ 3: Создать AppBundle структуру с LISP

# Создать структуру bundle

Get-ChildItem "C:\out\BtiPlugin.bundle\Contents"

ШАГ 4: Создать PackageContents.xml

$xml = @"
<?xml version="1.0" encoding="utf-8"?>
<ApplicationPackage SchemaVersion="1.0" AutodeskProduct="AutoCAD">
  <Components>
    <RuntimeRequirements OS="Win64" Platform="AutoCAD" />
    <ComponentEntry AppName="BtiPlugin" ModuleName="./Contents/BTI_InsertBasman.dll">
      <Commands GroupName="BTI_COMMANDS">
        <Command Global="BTI_INSERT_TEMPLATE" Local="BTI_INSERT_TEMPLATE" />
        <Command Global="BTI_APPLY" Local="BTI_APPLY" />
        <Command Global="BTI_CLEANUP" Local="BTI_CLEANUP" />
      </Commands>
    </ComponentEntry>
  </Components>
</ApplicationPackage>
"@

Set-Content "C:\out\BtiPlugin.bundle\PackageContents.xml" -Value $xml -Encoding UTF8

Write-Host "✅ PackageContents.xml создан"
