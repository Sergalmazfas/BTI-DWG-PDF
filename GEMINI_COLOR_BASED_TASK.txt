ШАГ 2: Создать AppBundle структуру (БЕЗ компиляции .NET)

# Создаем bundle только с LISP (работает без AutoCAD!)

Write-Host "`n📦 Содержимое bundle:"

ШАГ 3: Создать PackageContents.xml (упрощенный для LISP)

$xml = @"
<?xml version="1.0" encoding="utf-8"?>
<ApplicationPackage SchemaVersion="1.0" Version="1.0" AutodeskProduct="AutoCAD">
  <Components Description="BTI Color-Based Processor">
    <RuntimeRequirements OS="Win64" Platform="AutoCAD" />
  </Components>
</ApplicationPackage>
"@

Set-Content "C:\out\BtiPlugin.bundle\PackageContents.xml" -Value $xml -Encoding UTF8

Write-Host "✅ PackageContents.xml создан (LISP-only версия)"

ШАГ 4: Создать ZIP

$hash = Get-FileHash "C:\out\BtiPlugin.zip" -Algorithm SHA256

Write-Host "`n✅ ZIP создан:"
