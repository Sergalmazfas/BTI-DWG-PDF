ШАГ 1: Скачать LISP скрипты из GCS

Invoke-WebRequest -Uri "https://storage.googleapis.com/btibot-processed/scripts/BTI_APPLY.lsp" -OutFile "C:\BTI-DWG-PDF\scripts\BTI_APPLY.lsp"
Invoke-WebRequest -Uri "https://storage.googleapis.com/btibot-processed/scripts/BTI_CLEANUP.lsp" -OutFile "C:\BTI-DWG-PDF\scripts\BTI_CLEANUP.lsp"

═══════════════════════════════════════════════════════════════

ШАГ 2: Запустить build-windows.ps1 (компиляция .NET плагина)

cd C:\BTI-DWG-PDF
.\build-windows.ps1

Результат: C:\out\BTI_InsertBasman.dll (если AutoCAD установлен)

═══════════════════════════════════════════════════════════════

ШАГ 3: Создать AppBundle с LISP

New-Item -Path "C:\out\BtiPlugin.bundle\Contents" -ItemType Directory -Force

# Копировать файлы в bundle
Copy-Item "C:\out\BTI_InsertBasman.dll" -Destination "C:\out\BtiPlugin.bundle\Contents\" -ErrorAction SilentlyContinue
Copy-Item "C:\BTI-DWG-PDF\scripts\BTI_APPLY.lsp" -Destination "C:\out\BtiPlugin.bundle\Contents\"
Copy-Item "C:\BTI-DWG-PDF\scripts\BTI_CLEANUP.lsp" -Destination "C:\out\BtiPlugin.bundle\Contents\"
Copy-Item "C:\BTI-DWG-PDF\bti_basmanny_template.dwg" -Destination "C:\out\BtiPlugin.bundle\Contents\" -ErrorAction SilentlyContinue

═══════════════════════════════════════════════════════════════

ШАГ 4: Создать PackageContents.xml

$xml = @"
<?xml version="1.0" encoding="utf-8"?>
<ApplicationPackage SchemaVersion="1.0">
  <Components>
    <RuntimeRequirements OS="Win64" Platform="AutoCAD" />
    <ComponentEntry AppName="BtiPlugin" ModuleName="./Contents/BTI_InsertBasman.dll">
      <Commands GroupName="BTI">
        <Command Global="BTI_INSERT_TEMPLATE" Local="BTI_INSERT_TEMPLATE" />
        <Command Global="BTI_APPLY" Local="BTI_APPLY" />
        <Command Global="BTI_CLEANUP" Local="BTI_CLEANUP" />
      </Commands>
    </ComponentEntry>
  </Components>
</ApplicationPackage>
"@

Set-Content "C:\out\BtiPlugin.bundle\PackageContents.xml" -Value $xml

═══════════════════════════════════════════════════════════════

ШАГ 5: Создать ZIP архив

Compress-Archive -Path "C:\out\BtiPlugin.bundle" -DestinationPath "C:\out\BtiPlugin.zip" -Force

# Проверить
Get-Item C:\out\BtiPlugin.zip | Format-List Name, Length
Get-FileHash C:\out\BtiPlugin.zip -Algorithm SHA256

═══════════════════════════════════════════════════════════════

ШАГ 6: Загрузить в GCS

gsutil cp C:\out\BtiPlugin.zip gs://btibot-processed/appbundles/BtiPlugin_v1.zip

═══════════════════════════════════════════════════════════════

ШАГ 7: Зарегистрировать в Autodesk APS
